namespace GSM.Core.Source.Device.Logging

    /// <summary>
    /// Central logger for Device operations
    /// Provides structured logging with Serilog
    /// </summary>
    public class DeviceLogger : IDisposable
    {
        private static readonly Dictionary<string, DeviceLogger> _instances = new Dictionary<string, DeviceLogger>();
        private readonly ILogger _logger;
        private readonly string _deviceId;
        private bool _disposed = false;

        private DeviceLogger(string deviceId)
        {
            _deviceId = deviceId;

            // Serilog Configuration - Comprehensive logging
            _logger = new LoggerConfiguration()
                .MinimumLevel.Debug()
                // Text log file
                .WriteTo.File(
                    path: $"logs/device-{deviceId}-.txt",
                    outputTemplate: "{Timestamp:yyyy-MM-dd HH:mm:ss.fff zzz} [{Level:u3}] {Message:lj}{NewLine}{Exception}",
                    rollingInterval: RollingInterval.Day,
                    fileSizeLimitBytes: 10485760, // 10MB
                    retainedFileCountLimit: null) // Unlimited retention
                                                  // JSON structured log file
                .WriteTo.File(
    formatter: new Serilog.Formatting.Compact.CompactJsonFormatter(),
    path: $"logs/device-{deviceId}-structured-.json",
    rollingInterval: RollingInterval.Day,
    fileSizeLimitBytes: 10485760,
    retainedFileCountLimit: null)
                /*
                .WriteTo.File(
                    formatter: new Serilog.Formatting.Compact.CompactJsonFormatter(),
                    path: $"logs/device-{deviceId}-structured-.json",
                    rollingInterval: RollingInterval.Day,
                    fileSizeLimitBytes: 10485760,
                    retainedFileCountLimit: null)
                */
                // Enrich logs with context
                .Enrich.FromLogContext()
                .Enrich.WithProperty("DeviceId", deviceId)
                .Enrich.WithProperty("Timestamp", DateTime.UtcNow)
                .CreateLogger();

            // Log initialization
            _logger.Information("🟢 Logger initialized for Device: {DeviceId}", deviceId);
        }

        /// <summary>
        /// Get or create logger instance for device
        /// </summary>
        public static DeviceLogger GetInstance(string deviceId)
        {
            if (string.IsNullOrEmpty(deviceId))
                deviceId = "DEFAULT";

            if (!_instances.ContainsKey(deviceId))
            {
                _instances[deviceId] = new DeviceLogger(deviceId);
            }
            return _instances[deviceId];
        }

        // ================== OPERATION LIFECYCLE ==================

        /// <summary>
        /// Log operation start
        /// </summary>
        public LogContext OperationStart(string operationName, object parameters = null)
        {
            var correlationId = Guid.NewGuid().ToString("N").Substring(0, 12).ToUpper();
            var context = new LogContext
            {
                CorrelationId = correlationId,
                OperationName = operationName,
                StartTime = DateTime.UtcNow,
                Parameters = parameters
            };

            _logger.Information(
                "⏹️  OPERATION_START | Operation={Operation} | CorrelationId={CorrelationId} | Params={@Parameters}",
                operationName, correlationId, parameters ?? new { });

            return context;
        }

        /// <summary>
        /// Log operation successful completion
        /// </summary>
        public void OperationComplete(LogContext context, string status = "SUCCESS", object result = null)
        {
            if (context == null) return;

            var duration = DateTime.UtcNow - context.StartTime;

            _logger.Information(
                "✅ OPERATION_COMPLETE | Operation={Operation} | CorrelationId={CorrelationId} | Status={Status} | Duration={DurationMs}ms | Result={@Result}",
                context.OperationName, context.CorrelationId, status, duration.TotalMilliseconds, result ?? new { });
        }

        /// <summary>
        /// Log operation failure
        /// </summary>
        public void OperationFailed(LogContext context, Exception ex)
        {
            if (context == null) return;

            var duration = DateTime.UtcNow - context.StartTime;

            _logger.Error(ex,
                "❌ OPERATION_FAILED | Operation={Operation} | CorrelationId={CorrelationId} | Duration={DurationMs}ms | Error={ErrorMessage}",
                context.OperationName, context.CorrelationId, duration.TotalMilliseconds, ex?.Message ?? "Unknown");
        }

        // ================== ERROR LOGGING ==================

        /// <summary>
        /// Log error with code and message
        /// </summary>
        public void LogError(string errorCode, string errorMessage, string operation, Exception ex = null)
        {
            _logger.Error(ex,
                "🔴 ERROR | Code={ErrorCode} | Message={ErrorMessage} | Operation={Operation} | Time={Timestamp:yyyy-MM-dd HH:mm:ss.fff}",
                errorCode, errorMessage, operation, DateTime.UtcNow);
        }

        /// <summary>
        /// Log warning message
        /// </summary>
        public void LogWarning(string message, string operation)
        {
            _logger.Warning(
                "⚠️  WARNING | Message={Message} | Operation={Operation} | Time={Timestamp:yyyy-MM-dd HH:mm:ss.fff}",
                message, operation, DateTime.UtcNow);
        }

        // ================== DLL OPERATIONS ==================

        /// <summary>
        /// Log DLL method call
        /// </summary>
        public void LogDllCall(string methodName, Dictionary<string, object> parameters = null)
        {
            _logger.Debug(
                "🔗 DLL_CALL | Method={Method} | Parameters={@Parameters}",
                methodName, parameters ?? new Dictionary<string, object>());
        }

        /// <summary>
        /// Log DLL method result
        /// </summary>
        public void LogDllResult(string methodName, object result, long durationMs)
        {
            _logger.Debug(
                "🔗 DLL_RESULT | Method={Method} | Duration={DurationMs}ms | Result={@Result}",
                methodName, durationMs, result ?? new { });
        }

        // ================== STATE CHANGES ==================

        /// <summary>
        /// Log internal state change
        /// </summary>
        public void LogStateChange(string oldState, string newState, string reason)
        {
            _logger.Information(

        /// <summary>
        /// Log flag change
        /// </summary>
        public void LogFlagChange(string flagName, object oldValue, object newValue)
        {
            _logger.Debug(
                "🚩 FLAG_CHANGE | Flag={Flag} | From={OldValue} | To={NewValue}",
                flagName, oldValue, newValue);
        }

        public void Dispose()
        {
            if (_disposed) return;

            Log.CloseAndFlush();
            _disposed = true;
        }

using GSM.Core.Source.Common;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;

namespace GSM.Core.Source.Device.Logging
{
    /// <summary>
    /// Traces and records all errors in the device lifecycle
    /// Provides pattern analysis and error reporting
    /// </summary>
    public class ErrorTracer
    {
        private static ErrorTracer _instance;
        private readonly Queue<ErrorRecord> _errorHistory = new Queue<ErrorRecord>();
        private readonly DeviceLogger _logger;
        private const int MAX_HISTORY = 1000;

        public class ErrorRecord
        {
            public DateTime Timestamp { get; set; }
            public string ErrorCode { get; set; }
            public string ErrorMessage { get; set; }
            public string Operation { get; set; }
            public Exception Exception { get; set; }
            public int Layer { get; set; } // Controller=1, Workflow=2, Base=3, DLL=4
            public string LayerName =>
                Layer == 1 ? "Controller" :
                Layer == 2 ? "Workflow" :
                Layer == 3 ? "Base" :
                Layer == 4 ? "DLL" :
                "Unknown";
        }

        private ErrorTracer(DeviceLogger logger)
        {
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
        }

        /// <summary>
        /// Get or create singleton instance
        /// </summary>
        public static ErrorTracer GetInstance(DeviceLogger logger)
        {
            return _instance ?? (_instance = new ErrorTracer(logger));
        }

        /// <summary>
        /// Record an error occurrence
        /// </summary>
        public void RecordError(Device.ERROR_CODE code, string message, string operation, int layer = 1, Exception ex = null)
        {
            var record = new ErrorRecord
            {
                Timestamp = DateTime.UtcNow,
                ErrorCode = code.ToString(),
                ErrorMessage = message,
                Operation = operation,
                Layer = layer,
                Exception = ex
            };

            _errorHistory.Enqueue(record);

            // Maintain maximum size
            if (_errorHistory.Count > MAX_HISTORY)
                _errorHistory.Dequeue();

            // Log the error
            _logger?.LogError(code.ToString(), message, $"{operation} [{record.LayerName}]", ex);
        }

        /// <summary>
        /// Get error patterns grouped by code
        /// </summary>
        public List<(string Code, int Count, DateTime FirstOccurrence, DateTime LastOccurrence, string CommonMessage)> GetErrorPatterns()
        {
            return _errorHistory
                .GroupBy(e => e.ErrorCode)
                .Select(g => (
                    g.Key,
                    g.Count(),
                    g.Min(x => x.Timestamp),
                    g.Max(x => x.Timestamp),
                    g.OrderByDescending(x => x.Timestamp).First().ErrorMessage))
                .OrderByDescending(x => x.Item2)
                .ToList();
        }

        /// <summary>
        /// Get recent errors
        /// </summary>
        public List<ErrorRecord> GetRecentErrors(int count = 20)
        {
            return _errorHistory
                .OrderByDescending(e => e.Timestamp)
                .Take(count)
                .ToList();
        }

        /// <summary>
        /// Get errors by operation
        /// </summary>
        public List<ErrorRecord> GetErrorsByOperation(string operation)
        {
            return _errorHistory
                .Where(e => e.Operation == operation)
                .OrderByDescending(e => e.Timestamp)
                .ToList();
        }

        /// <summary>
        /// Get errors by error code
        /// </summary>
        public List<ErrorRecord> GetErrorsByCode(string errorCode)
        {
            return _errorHistory
                .Where(e => e.ErrorCode == errorCode)
                .OrderByDescending(e => e.Timestamp)
                .ToList();
        }

        /// <summary>
        /// Get critical errors (last 24 hours)
        /// </summary>
        public List<ErrorRecord> GetCriticalErrors()
        {
            var twentyFourHoursAgo = DateTime.UtcNow.AddHours(-24);
            return _errorHistory
                .Where(e => e.Timestamp >= twentyFourHoursAgo)
                .OrderByDescending(e => e.Timestamp)
                .ToList();
        }

        /// <summary>
        /// Get total error count
        /// </summary>
        public int GetTotalErrorCount()
        {
            return _errorHistory.Count;
        }

        /// <summary>
        /// Generate comprehensive error report
        /// </summary>
        public string GenerateErrorReport()
        {
            var patterns = GetErrorPatterns();
            var sb = new StringBuilder();

            sb.AppendLine("════════════════════════════════════════════════════════════");
            sb.AppendLine("                      ERROR REPORT                          ");
            sb.AppendLine("════════════════════════════════════════════════════════════");
            sb.AppendLine($"Generated: {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss.fff}");
            sb.AppendLine($"Total Errors: {_errorHistory.Count}");
            sb.AppendLine($"Unique Error Codes: {patterns.Count}");
            sb.AppendLine();

            sb.AppendLine("Top 10 Error Patterns:");
            int index = 1;
            foreach (var (code, count, firstOcc, lastOcc, lastMsg) in patterns.Take(10))
            {
                sb.AppendLine($"  {index}. [{code}] - Occurred {count} times");
                sb.AppendLine($"     ├─ First: {firstOcc:yyyy-MM-dd HH:mm:ss}");
                sb.AppendLine($"     ├─ Last:  {lastOcc:yyyy-MM-dd HH:mm:ss}");
                sb.AppendLine($"     └─ Message: {lastMsg}");
                sb.AppendLine();
                index++;
            }

            sb.AppendLine("Recent 5 Errors:");
            var recent = GetRecentErrors(5);
            for (int i = 0; i < recent.Count; i++)
            {
                var err = recent[i];
                sb.AppendLine($"  {i + 1}. [{err.ErrorCode}] at {err.Timestamp:yyyy-MM-dd HH:mm:ss.fff}");
                sb.AppendLine($"     ├─ Operation: {err.Operation}");
                sb.AppendLine($"     ├─ Layer: {err.LayerName}");
                sb.AppendLine($"     └─ Message: {err.ErrorMessage}");
                sb.AppendLine();
            }

            sb.AppendLine("════════════════════════════════════════════════════════════");

            return sb.ToString();
        }

        /// <summary>
        /// Clear error history
        /// </summary>
        public void ClearHistory()
        {
            _errorHistory.Clear();
        }

        /// <summary>
        /// Export error history as CSV
        /// </summary>
        public string ExportAsCSV()
        {
            var sb = new StringBuilder();
            sb.AppendLine("Timestamp,ErrorCode,Operation,Layer,Message");

            foreach (var error in _errorHistory.OrderBy(e => e.Timestamp))
            {
                sb.AppendLine($"\"{error.Timestamp:yyyy-MM-dd HH:mm:ss.fff}\",\"{error.ErrorCode}\",\"{error.Operation}\",\"{error.LayerName}\",\"{error.ErrorMessage}\"");
            }

            return sb.ToString();
        }
    }
}

using System;

namespace GSM.Core.Source.Device.Logging
{
    /// <summary>
    /// Extension methods for automatic operation logging
    /// Provides zero-code-change logging capability
    /// </summary>
    public static class LoggingExtensions
    {
        /// <summary>
        /// Wraps a returning operation with automatic logging
        /// Usage: return device.LogOperation(() => operation(), "OperationName");
        /// </summary>
        public static T LogOperation<T>(this Device device,
                                       Func<T> operation,
                                       string operationName)
        {
            if (string.IsNullOrEmpty(operationName))
                throw new ArgumentException("Operation name cannot be null or empty", nameof(operationName));

            var logger = DeviceLogger.GetInstance(device?.DllFileName ?? "Device");
            var context = logger.OperationStart(operationName);

            try
            {
                var result = operation.Invoke();
                logger.OperationComplete(context, "SUCCESS", new { Result = result });
                return result;
            }
            catch (Exception ex)
            {
                logger.OperationFailed(context, ex);
                throw;
            }
        }

        /// <summary>
        /// Wraps a void operation with automatic logging
        /// Usage: device.LogOperation(() => DoSomething(), "OperationName");
        /// </summary>
        public static void LogOperation(this Device device,
                                       Action operation,
                                       string operationName)
        {
            if (string.IsNullOrEmpty(operationName))
                throw new ArgumentException("Operation name cannot be null or empty", nameof(operationName));

            var logger = DeviceLogger.GetInstance(device?.DllFileName ?? "Device");
            var context = logger.OperationStart(operationName);

            try
            {
                operation.Invoke();
                logger.OperationComplete(context, "SUCCESS");
            }
            catch (Exception ex)
            {
                logger.OperationFailed(context, ex);
                throw;
            }
        }

        /// <summary>
        /// Get logger instance for the device
        /// </summary>
        public static DeviceLogger GetLogger(this Device device)
        {
            return DeviceLogger.GetInstance(device?.DllFileName ?? "Device");
        }

        /// <summary>
        /// Get error tracer instance for the device
        /// </summary>
        public static ErrorTracer GetErrorTracer(this Device device)
        {
            var logger = GetLogger(device);
            return ErrorTracer.GetInstance(logger);
        }
    }

using System;
using System.IO;

namespace GSM.Core.Source.Device.Logging
{
    /// <summary>
    /// Helper class for initializing logging infrastructure
    /// </summary>
    public static class LoggingInitializer
    {
        private static bool _initialized = false;

        /// <summary>
        /// Initialize logging system
        /// Call this once at application startup
        /// </summary>
        public static void Initialize()
        {
            if (_initialized) return;

            try
            {
                // Create logs directory if it doesn't exist
                string logsDirectory = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "logs");
                if (!Directory.Exists(logsDirectory))
                {
                    Directory.CreateDirectory(logsDirectory);
                }

                _initialized = true;
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine($"Failed to initialize logging: {ex.Message}");
            }
        }

        /// <summary>
        /// Check if logging is initialized
        /// </summary>
        public static bool IsInitialized => _initialized;
    }
}

## 3️⃣ پلن #2: رصد عملکرد (Performance Monitoring)

### 🎯 هدف

### 🏗️ معماری پیشنهادی

```csharp
public class PerformanceTracker
{
	private Dictionary<string, OperationMetrics> _metrics = new();

	public class OperationMetrics
	{
		public string OperationName { get; set; }
		public long MinDuration { get; set; }
		public long MaxDuration { get; set; }
		public long AverageDuration { get; set; }
		public int ExecutionCount { get; set; }
		public int FailureCount { get; set; }
		public double SuccessRate { get; set; }
		public List<long> RecentDurations { get; set; } = new();
	}

	public void RecordOperation(string operation, long durationMs, bool success)
	{
		if (!_metrics.ContainsKey(operation))
			_metrics[operation] = new OperationMetrics { OperationName = operation };

		var metric = _metrics[operation];

		metric.ExecutionCount++;
		if (!success) metric.FailureCount++;
		metric.SuccessRate = (metric.ExecutionCount - metric.FailureCount) 
						   / (double)metric.ExecutionCount * 100;

		metric.MinDuration = Math.Min(metric.MinDuration, durationMs);
		metric.MaxDuration = Math.Max(metric.MaxDuration, durationMs);
		metric.RecentDurations.Add(durationMs);

		// Keep only last 100 durations
		if (metric.RecentDurations.Count > 100)
			metric.RecentDurations.RemoveAt(0);

		metric.AverageDuration = (long)metric.RecentDurations.Average();
	}

	public Dictionary<string, OperationMetrics> GetMetrics() => _metrics;

	public string GenerateReport()
	{
		var sb = new StringBuilder();
		foreach (var metric in _metrics.Values)
		{
			sb.AppendLine($"Operation: {metric.OperationName}");
			sb.AppendLine($"  Executions: {metric.ExecutionCount}");
			sb.AppendLine($"  Success Rate: {metric.SuccessRate:F2}%");
			sb.AppendLine($"  Min Duration: {metric.MinDuration}ms");
			sb.AppendLine($"  Max Duration: {metric.MaxDuration}ms");
			sb.AppendLine($"  Avg Duration: {metric.AverageDuration}ms");
			sb.AppendLine();
		}
		return sb.ToString();
	}
}
```

```csharp
public class ErrorTracing
{
	public class ErrorContext
	{
		public string CorrelationId { get; set; }
		public string OperationName { get; set; }
		public ERROR_CODE ErrorCode { get; set; }
		public string ErrorMessage { get; set; }
		public int Layer { get; set; } // Controller=1, Workflow=2, Base=3, DLL=4
		public DateTime Timestamp { get; set; }
		public Stack<string> CallStack { get; set; }
		public Dictionary<string, object> DeviceState { get; set; }
		public Exception Exception { get; set; }
	}

	private Queue<ErrorContext> _errorHistory = new(maxSize: 1000);

	public void RecordError(ERROR_CODE code, string message, string operation, 
						   int layer, Exception ex = null)
	{
		var context = new ErrorContext
		{
			CorrelationId = Guid.NewGuid().ToString(),
			OperationName = operation,
			ErrorCode = code,
			ErrorMessage = message,
			Layer = layer,
			Timestamp = DateTime.UtcNow,
			CallStack = new Stack<string>(new[] 
			{ 
				"Controller.DeviceCountGet -> Workflow.DeviceCountGetWorkflow -> Execute -> Base"
			}),
			DeviceState = CaptureDeviceState(),
			Exception = ex
		};

		_errorHistory.Enqueue(context);
		LogErrorPattern(context);
	}

	// Pattern Analysis
	public class ErrorPattern
	{
		public ERROR_CODE Code { get; set; }
		public int Frequency { get; set; }
		public DateTime FirstOccurrence { get; set; }
		public DateTime LastOccurrence { get; set; }
		public double FrequencyPerHour { get; set; }
		public string CommonMessage { get; set; }
	}

	public List<ErrorPattern> AnalyzeErrors(TimeSpan period)
	{
		var patterns = new Dictionary<ERROR_CODE, ErrorPattern>();

		foreach (var error in _errorHistory.Where(e => DateTime.UtcNow - e.Timestamp < period))
		{
			if (!patterns.ContainsKey(error.ErrorCode))
				patterns[error.ErrorCode] = new ErrorPattern 
				{ 
					Code = error.ErrorCode,
					FirstOccurrence = error.Timestamp
				};

			var pattern = patterns[error.ErrorCode];
			pattern.Frequency++;
			pattern.LastOccurrence = error.Timestamp;
			pattern.CommonMessage = error.ErrorMessage;
		}

		return patterns.Values.ToList();
	}
}
```

```csharp
public class StateMonitor
{
	public class StateSnapshot

		public int DeviceIndex { get; set; }
		public Dictionary<string, object> RuntimeFlags { get; set; }
		public Dictionary<string, int> RuntimeCounters { get; set; }
	}

	private List<StateSnapshot> _stateHistory = new();
	private StateSnapshot _lastState;

	public void CaptureState(Device device)
	{
		var snapshot = new StateSnapshot
		{
			Timestamp = DateTime.UtcNow,
			InternalStatus = device.InternalStatus,
			Busy = device.Busy,
			LastErrorCode = device.ErrorCode,
			DeviceCount = device.DeviceCount,
			DeviceIndex = device.DeviceIndex,

			RuntimeCounters = CaptureCounters(device)
		};

		_stateHistory.Add(snapshot);
		DetectStateAnomalies(snapshot);
		_lastState = snapshot;
	}

	public void DetectStateAnomalies(StateSnapshot current)
	{
		if (_lastState == null) return;

		// Anomaly 1: Unexpected state transition
		if (IsInvalidStateTransition(current.InternalStatus, _lastState.InternalStatus))
		{
			LogAnomaly($"Invalid state transition: {_lastState.InternalStatus} -> {current.InternalStatus}");
		}

		// Anomaly 2: Stuck in error state
		if (current.LastErrorCode != ERROR_CODE.NO_ERROR && 
			_lastState.LastErrorCode == current.LastErrorCode &&
			(current.Timestamp - _lastState.Timestamp).TotalSeconds > 5)
		{
			LogAnomaly($"Stuck in error: {current.LastErrorCode}");
		}

		// Anomaly 3: Busy flag stuck
		if (current.Busy == 1 && _lastState.Busy == 1 &&
			(current.Timestamp - _lastState.Timestamp).TotalSeconds > 30)
		{
			LogAnomaly("Busy flag stuck for more than 30 seconds");
		}
	}

	public StateReport GenerateStateReport()
	{
		return new StateReport
		{
			TotalStateChanges = _stateHistory.Count,
			CurrentState = _lastState?.InternalStatus.ToString(),
			StateChangeFrequency = CalculateFrequency(),
			AnomalyCount = _anomalies.Count,
			AnomalyDetails = _anomalies.Take(10).ToList()
		};
	}
}
```

```csharp
public class DllOperationTracker
{
	public class DllCallRecord
	{
		public string MethodName { get; set; }
		public Dictionary<string, object> Parameters { get; set; }
		public object Result { get; set; }
		public long DurationMs { get; set; }
		public DateTime Timestamp { get; set; }
		public bool Success { get; set; }
		public string ErrorInfo { get; set; }
	}

	private Queue<DllCallRecord> _callRecords = new(maxSize: 5000);

	public void RecordDllCall(string method, Dictionary<string, object> parameters,
							 object result, long duration, bool success, string error = null)
	{
		var record = new DllCallRecord
		{
			MethodName = method,
			Parameters = parameters,
			Result = result,
			DurationMs = duration,
			Timestamp = DateTime.UtcNow,
			Success = success,
			ErrorInfo = error
		};

		_callRecords.Enqueue(record);
	}

	public DllHealthReport GenerateHealthReport()
	{
		return new DllHealthReport
		{
			TotalCalls = _callRecords.Count,
			SuccessCalls = _callRecords.Count(c => c.Success),
			FailedCalls = _callRecords.Count(c => !c.Success),
			SuccessRate = (_callRecords.Count(c => c.Success) / (double)_callRecords.Count) * 100,
			AverageDuration = _callRecords.Average(c => c.DurationMs),
			MostCalledMethod = _callRecords
				.GroupBy(c => c.MethodName)
				.OrderByDescending(g => g.Count())
				.First().Key,
			SlowestMethod = _callRecords
				.GroupBy(c => c.MethodName)
				.OrderByDescending(g => g.Average(c => c.DurationMs))
				.First().Key,
			RecentErrors = _callRecords
				.Where(c => !c.Success)
				.OrderByDescending(c => c.Timestamp)
				.Take(20)
				.ToList()
		};
	}
}

public class DllHealthReport
{
	public int TotalCalls { get; set; }
	public int SuccessCalls { get; set; }
	public int FailedCalls { get; set; }
	public double SuccessRate { get; set; }
	public double AverageDuration { get; set; }
	public string MostCalledMethod { get; set; }
	public string SlowestMethod { get; set; }
	public List<DllOperationTracker.DllCallRecord> RecentErrors { get; set; }
}
```

```csharp
public class DeviceAnalyticsEngine
{
	private readonly PerformanceTracker _performanceTracker;
	private readonly ErrorTracing _errorTracing;
	private readonly StateMonitor _stateMonitor;
	private readonly DllOperationTracker _dllTracker;

	public class AnalyticsSnapshot
	{
		public DateTime Timestamp { get; set; }

		// Health Metrics
		public double OverallHealthScore { get; set; } // 0-100
		public string HealthStatus { get; set; } // Healthy, Warning, Critical

		// Performance Summary
		public double AverageOperationDuration { get; set; }
		public double OperationSuccessRate { get; set; }

		// Error Summary
		public int ErrorCount24h { get; set; }
		public int CriticalErrorCount { get; set; }
		public List<ErrorPattern> TopErrors { get; set; }

		// State Summary
		public string CurrentInternalStatus { get; set; }
		public int StateChangeCount24h { get; set; }

		// DLL Summary
		public double DllSuccessRate { get; set; }
		public double AvgDllDuration { get; set; }

		// Recommendations
		public List<string> Recommendations { get; set; }
	}

	public AnalyticsSnapshot GenerateSnapshot()
	{
		var perfMetrics = _performanceTracker.GetMetrics();
		var errors = _errorTracing.AnalyzeErrors(TimeSpan.FromHours(24));
		var stateReport = _stateMonitor.GenerateStateReport();
		var dllReport = _dllTracker.GenerateHealthReport();

		var snapshot = new AnalyticsSnapshot
		{
			Timestamp = DateTime.UtcNow,
			OverallHealthScore = CalculateHealthScore(perfMetrics, errors, dllReport),
			HealthStatus = DetermineHealthStatus(perfMetrics, errors, dllReport),
			AverageOperationDuration = perfMetrics.Values.Average(m => m.AverageDuration),
			OperationSuccessRate = perfMetrics.Values.Average(m => m.SuccessRate),
			ErrorCount24h = errors.Sum(e => e.Frequency),
			CriticalErrorCount = errors.Count(e => e.Code == ERROR_CODE.DLL_WRAPPER || 
													e.Code == ERROR_CODE.EXECUTE_TIMEOUT),
			TopErrors = errors.OrderByDescending(e => e.Frequency).Take(5).ToList(),
			Recommendations = GenerateRecommendations(perfMetrics, errors, dllReport)
		};

		return snapshot;
	}

	private List<string> GenerateRecommendations(Dictionary<string, OperationMetrics> perf,
												 List<ErrorPattern> errors,
												 DllHealthReport dllReport)
	{
		var recommendations = new List<string>();

		// Performance recommendations
		var slowOps = perf.Where(p => p.Value.AverageDuration > 200).ToList();
		if (slowOps.Any())
			recommendations.Add($"⚠️ {slowOps.Count} operations are slow (>{200}ms avg). Consider optimization.");

		// Error recommendations
		var highErrorOps = perf.Where(p => p.Value.SuccessRate < 95).ToList();
		if (highErrorOps.Any())
			recommendations.Add($"🔴 {highErrorOps.Count} operations have low success rates. Investigate immediately.");

		// DLL recommendations
		if (dllReport.SuccessRate < 99)
			recommendations.Add($"⚠️ DLL success rate is {dllReport.SuccessRate:F2}%. Check DLL stability.");

		// Resource recommendations
		if (dllReport.AverageDuration > 100)
			recommendations.Add("⚠️ Average DLL call duration is high. Check system resources.");

		if (!recommendations.Any())
			recommendations.Add("✅ System is running optimally. No issues detected.");

		return recommendations;
	}
}
```

```csharp
private FUNC_RESULT DllWrapperHasError(FUNC_RESULT funcResult)
{
	try
	{
		if (this._dllWrapper.ErrorCode == DllWrapper.ERROR_CODE.NO_ERROR)
			return funcResult;
		else
		{
			this.DllWrapperErrorCheckSetValue(this._dllWrapper.ErrorMessage);

			// ✅ اضافه شد - Log DLL Error
			if (this._logger != null)
			{
				this._logger.LogDllResult("DllWrapper", "ERROR", 0);
			}

			return FUNC_RESULT.MAJOR_ERROR;
		}
	}
	catch (Exception ex)
	{
		this.DllWrapperErrorCheckSetValue(ex.Message);

		// ✅ اضافه شد - Log Exception
		if (this._logger != null)
		{
			this._logger.LogError("DLL_ERROR", ex.Message, "DllWrapperHasError", ex);
		}

		return FUNC_RESULT.MAJOR_ERROR;
	}
}
```

```csharp
var logger = DeviceLogger.GetInstance("Device");

var context = logger.OperationStart("MyOperation", new { param = "value" });
logger.OperationComplete(context, "SUCCESS", new { result = data });
logger.OperationFailed(context, exception);

// Error logging
logger.LogError("ERROR_CODE", "message", "operation", exception);
logger.LogWarning("warning message", "operation");

// DLL logging
logger.LogDllCall("DllMethod", new { param = "value" });
logger.LogDllResult("DllMethod", result, durationMs);

// State logging
logger.LogStateChange("INVALID", "BASIC_LEVEL", "reason");
logger.LogFlagChange("BusyFlag", 0, 1);
```

```csharp
// After (Option A - Simple)
public bool DeviceCountGet()
{
	var logger = this._logger ?? DeviceLogger.GetInstance("Device");
	var ctx = logger.OperationStart("DeviceCountGet");
	try
	{
		int count = 0;
		bool result = this.DeviceCountGet(ref count, true);
		logger.OperationComplete(ctx, "SUCCESS");
		return result;
	}
	catch (Exception ex)
	{
		logger.OperationFailed(ctx, ex);
		throw;
	}
}

// After (Option B - Using Extension)
public bool DeviceCountGet()
{
	return this.LogOperation(

		"DeviceCountGet"
	);
}
```

```csharp
using Serilog;
using Serilog.Core;
using Serilog.Events;
using System;
using System.Collections.Generic;

namespace GSM.Core.Source.Device.Logging
{
	public class DeviceLogger : IDisposable
	{
		private static readonly Dictionary<string, DeviceLogger> _instances = new();
		private readonly ILogger _logger;
		private readonly string _deviceId;

		private DeviceLogger(string deviceId)
		{
			_deviceId = deviceId;

			// Serilog Configuration
			_logger = new LoggerConfiguration()
				.MinimumLevel.Debug()
				.WriteTo.File(
					path: $"logs/device-{deviceId}-.txt",
					outputTemplate: "{Timestamp:yyyy-MM-dd HH:mm:ss.fff zzz} [{Level:u3}] {Message:lj}{NewLine}{Exception}",
					rollingInterval: RollingInterval.Day,
					fileSizeLimitBytes: 10485760, // 10MB
					retainedFileCountLimit: null) // نامحدود
				.WriteTo.File(
					formatter: new CompactJsonFormatter(),
					path: $"logs/device-{deviceId}-structured-.json",
					rollingInterval: RollingInterval.Day,
					fileSizeLimitBytes: 10485760,
					retainedFileCountLimit: null)
				.Enrich.FromLogContext()
				.Enrich.WithProperty("DeviceId", deviceId)
				.CreateLogger();
		}

		public static DeviceLogger GetInstance(string deviceId)
		{
			if (!_instances.ContainsKey(deviceId))
			{
				_instances[deviceId] = new DeviceLogger(deviceId);
			}
			return _instances[deviceId];
		}

		// ===== Operation Lifecycle =====
		public LogContext OperationStart(string operationName, object parameters = null)
		{
			var correlationId = Guid.NewGuid().ToString("N").Substring(0, 12);
			var context = new LogContext
			{
				CorrelationId = correlationId,
				OperationName = operationName,
				StartTime = DateTime.UtcNow,
				Parameters = parameters
			};

			_logger.Information(
				"⏹️ Operation START | Operation={Operation} | CorrelationId={CorrelationId} | Params={@Params}",
				operationName, correlationId, parameters ?? new { });

			return context;
		}

		public void OperationComplete(LogContext context, string status, object result = null)
		{
			var duration = DateTime.UtcNow - context.StartTime;

			_logger.Information(
				"✅ Operation COMPLETE | Operation={Operation} | CorrelationId={CorrelationId} | Status={Status} | Duration={DurationMs}ms | Result={@Result}",
				context.OperationName, context.CorrelationId, status, duration.TotalMilliseconds, result ?? new { });
		}

		public void OperationFailed(LogContext context, Exception ex)
		{
			var duration = DateTime.UtcNow - context.StartTime;

			_logger.Error(ex,
				"❌ Operation FAILED | Operation={Operation} | CorrelationId={CorrelationId} | Duration={DurationMs}ms | Error={ErrorMessage}",
				context.OperationName, context.CorrelationId, duration.TotalMilliseconds, ex.Message);
		}

		// ===== Error Logging =====
		public void LogError(string errorCode, string errorMessage, string operation, Exception ex = null)
		{
			_logger.Error(ex,
				"🔴 ERROR | Code={ErrorCode} | Message={ErrorMessage} | Operation={Operation}",
				errorCode, errorMessage, operation);
		}

		public void LogWarning(string message, string operation)
		{
			_logger.Warning(
				"⚠️ WARNING | Message={Message} | Operation={Operation}",
				message, operation);
		}

		// ===== DLL Logging =====
		public void LogDllCall(string methodName, Dictionary<string, object> parameters)
		{
			_logger.Debug(
				"🔗 DLL CALL | Method={Method} | Params={@Params}",
				methodName, parameters ?? new { });
		}

		public void LogDllResult(string methodName, object result, long durationMs)
		{
			_logger.Debug(
				"🔗 DLL RESULT | Method={Method} | Duration={DurationMs}ms | Result={@Result}",
				methodName, durationMs, result ?? new { });
		}

		// ===== State Changes =====
		public void LogStateChange(string oldState, string newState, string reason)
		{
			_logger.Information(
				"🔄 STATE CHANGE | From={OldState} | To={NewState} | Reason={Reason}",
				oldState, newState, reason);
		}

		public void Dispose()
		{
			Log.CloseAndFlush();
		}
	}

	// JSON Formatter for compact logs
	public class CompactJsonFormatter : ITextFormatter
	{
		public void Format(LogEvent logEvent, TextWriter output)
		{
			var json = new
			{
				Timestamp = logEvent.Timestamp.DateTime,
				Level = logEvent.Level.ToString(),
				Message = logEvent.RenderMessage(),
				Properties = logEvent.Properties,
				Exception = logEvent.Exception?.ToString()
			};

			output.WriteLine(System.Text.Json.JsonSerializer.Serialize(json));
		}
	}
}
```

```csharp
using GSM.Core.Source.Common;
using System;
using System.Collections.Generic;
using System.Linq;

namespace GSM.Core.Source.Device.Logging
{
	public class ErrorTracer
	{
		private static ErrorTracer _instance;
		private readonly Queue<ErrorRecord> _errorHistory = new();
		private readonly DeviceLogger _logger;
		private const int MAX_HISTORY = 1000;

		private ErrorTracer(DeviceLogger logger)
		{
			_logger = logger;
		}

		public static ErrorTracer GetInstance(DeviceLogger logger)
		{
			return _instance ?? (_instance = new ErrorTracer(logger));
		}

		public class ErrorRecord
		{
			public DateTime Timestamp { get; set; }
			public string ErrorCode { get; set; }
			public string ErrorMessage { get; set; }
			public string Operation { get; set; }
			public Exception Exception { get; set; }
			public int Layer { get; set; } // Controller=1, Workflow=2, Base=3, DLL=4
		}

		public void RecordError(ERROR_CODE code, string message, string operation, int layer = 1, Exception ex = null)
		{
			var record = new ErrorRecord
			{
				Timestamp = DateTime.UtcNow,
				ErrorCode = code.ToString(),
				ErrorMessage = message,
				Operation = operation,
				Layer = layer,
				Exception = ex
			};

			_errorHistory.Enqueue(record);

			// محدود کردن اندازه تاریخچه
			if (_errorHistory.Count > MAX_HISTORY)
				_errorHistory.Dequeue();

			// لاگ کردن
			_logger.LogError(code.ToString(), message, operation, ex);
		}

		public List<(string Code, int Count, string LastMessage)> GetErrorPatterns()
		{
			return _errorHistory
				.GroupBy(e => e.ErrorCode)
				.Select(g => (g.Key, g.Count(), g.OrderByDescending(x => x.Timestamp).First().ErrorMessage))
				.OrderByDescending(x => x.Item2)
				.ToList();
		}

		public List<ErrorRecord> GetRecentErrors(int count = 20)
		{
			return _errorHistory
				.OrderByDescending(e => e.Timestamp)
				.Take(count)
				.ToList();
		}

		public string GenerateErrorReport()
		{
			var patterns = GetErrorPatterns();
			var sb = new System.Text.StringBuilder();

			sb.AppendLine("════════════════════════════════════════════");
			sb.AppendLine("              ERROR REPORT                   ");
			sb.AppendLine("════════════════════════════════════════════");
			sb.AppendLine($"Total Errors: {_errorHistory.Count}");
			sb.AppendLine();

			sb.AppendLine("Top Errors:");
			foreach (var (code, count, lastMsg) in patterns.Take(10))
			{
				sb.AppendLine($"  ├─ {code}: {count} times");
				sb.AppendLine($"  │  └─ Last: {lastMsg}");
			}

			sb.AppendLine();
			sb.AppendLine("════════════════════════════════════════════");

			return sb.ToString();
		}
	}
}
```

```csharp
// در Constructor
public Device(string DllFileName)
{
	this.ErrorReset();
	this._dllFileName = DllFileName;

	// ✅ یک خط اضافه: Initialize Logger
	this._logger = DeviceLogger.GetInstance(DllFileName);
	this._errorTracer = ErrorTracer.GetInstance(this._logger);

	bool result = true;
	if (result) { result = this.DllLoad(); }
	if (result) { result = this.DllLoadCheck(); }

	this.ConstructorStatusSet(result);
}

// یک خط در ErrorSet() موجود:
private bool ErrorSet(ERROR_CODE ErrorCode, string ErrorMessage)
{
	this._state.ErrorCode = ErrorCode;
	this._state.ErrorMessage = ErrorMessage;

	// ✅ یک خط اضافه: Log the error
	this._errorTracer?.RecordError(ErrorCode, ErrorMessage, "Unknown");

	return false;
}

// یک خط دیگر اختیاری در DllWrapperHasError():
private FUNC_RESULT DllWrapperHasError(FUNC_RESULT funcResult)
{
	try
	{
		if (this._dllWrapper.ErrorCode == DllWrapper.ERROR_CODE.NO_ERROR)
			return funcResult;
		else
		{
			this.DllWrapperErrorCheckSetValue(this._dllWrapper.ErrorMessage);
			// ✅ لاگ DLL خطا
			this._logger?.LogDllResult("DllWrapper", "ERROR", 0);
			return FUNC_RESULT.MAJOR_ERROR;
		}
	}
	catch (Exception ex)
	{
		this.DllWrapperErrorCheckSetValue(ex.Message);
		this._logger?.LogError("DLL_ERROR", ex.Message, "DllWrapperHasError");
		return FUNC_RESULT.MAJOR_ERROR;
	}
}
```
